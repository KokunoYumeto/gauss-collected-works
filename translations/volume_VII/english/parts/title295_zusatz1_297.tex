\pdfbookmark[1]{1. On Art. 7.}{zusatz-001}[1. On Art. 7.]
\medskip

Approximate formulas for the maximum equation of centre.
\end{center}
\[
e=\sin\varphi
\]
\[
\begin{aligned}
\mathrm{I.}\quad &[v-M=]\ \varphi+e+\frac{3}{8}(\varphi-e),\\
&\text{gives: }2e+\frac{11}{48}e^3+\frac{33}{320}e^5,\\
&\text{Error: }-\frac{71}{5120}e^5.
\end{aligned}
\]
\[
\begin{aligned}
\mathrm{II.}\quad &[v-M=]\ e+2\operatorname{arc}\sin\BareRoot{(\sin\frac{1}{2}\varphi\operatorname{tang}\frac{1}{2}\varphi)},\\
&\text{gives: }2e+\frac{11}{48}e^3+\frac{297}{2560}e^5;\\
&\text{Error: }-\frac{5}{5120}e^5.
\end{aligned}
\]
