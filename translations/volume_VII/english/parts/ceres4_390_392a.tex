\begin{center}\pdfbookmark[2]{4. Calculation of the coefficients}{ceres-004}[4.]\end{center}
\noindent\hspace*{\parindent}[The coefficients \(A^{(i)}, B^{(i)}, C^{(i)}\) and so on are calculated as follows:]

Set
\[
\frac{1}{\BareRoot{(aa+a'a'-2aa'\cos w)}}=\tfrac12P^{(0)}+P'\cos w+P''\cos2w+\text{etc.}
\]
\[
\frac{1}{\BareRoot{(aa+a'a'-2aa'\cos w)}^{\frac32}}=\tfrac12Q^{(0)}+Q'\cos w+Q''\cos2w+\text{etc.}
\]
\noindent[and then, by analogy with the expansions given by \textsc{Laplace}, loc. cit., pp. 268—269, we have the recurrence formulas:]
\[
0=(i-\tfrac12)P^{(i-1)}-i\left(\frac a{a'}+\frac{a'}a\right)P^{(i)}+(i+\tfrac12)P^{(i+1)}
\]
\[
0=(i+\tfrac12)Q^{(i-1)}-i\left(\frac a{a'}+\frac{a'}a\right)Q^{(i)}+(i-\tfrac12)Q^{(i+1)}
\]
\[
\tfrac12\left(\frac a{a'}-\frac{a'}a\right)^2Q^{(i)}=\left(\frac a{a'}+\frac{a'}a\right)(i+\tfrac12)P^{(i)}-2(i+\tfrac12)P^{(i+1)},
\]
\noindent[from which all \(P^{(i)}\) and \(Q^{(i)}\) can be calculated once the first two coefficients \(P^{(0)}\) and \(P'\) have been found. A special method yields these more quickly than the usual series given by \textsc{Laplace}, loc. cit., pp. 271—272:]

Set
\[
\tfrac12(a+a')=x,\qquad\pm\tfrac12(a-a')=y
\]
\noindent[where in our case, \(a<a'\), the lower sign is to be taken], and form the sequences
\[
\begin{gathered}
x,\ x',\ x''\quad\text{etc.}\\
y,\ y',\ y''\quad\text{etc.}\\
z,\ z',\ z''\quad\text{etc.}
\end{gathered}
\]
\noindent as follows:
\[
\begin{aligned}
x'&=\tfrac12(x+y),&y'&\overset{\scriptscriptstyle\boldsymbol{.}}{=}\BareRoot{(xy)},\\
x''&=\tfrac12(x'+y'),&y''&=\BareRoot{(x'y')}\\
&\text{etc.}&&\text{etc.}
\end{aligned}
\]
\[
z=xx-yy=aa',
\]
\[
\begin{aligned}
z'&=x'x'-y'y'=\left(\frac z{4x'}\right)^2\\
&=\tfrac14aa\ \text{or}\ \tfrac14a'a',\ \text{according as}\ a<\text{or}>a',
\end{aligned}
\]
\[
\begin{gathered}
z''=x''x''-y''y''=\left(\frac{z'}{4x''}\right)^2\\
[\text{etc.}].
\end{gathered}
\]

\hypertarget{native-391}{}Then
\[
P^{(0)}=\frac1{x^{(\infty)}}=\frac1{y^{(\infty)}}\mathpunct{\text{\raisebox{.45ex}{,}}}
\]
\refstepcounter{footnote}
\[
\begin{aligned}
P'&=\frac{2P^{(0)}}z\left(z'+2z''+4z'''+8z^{\mathrm{IV}}+\text{etc.}\right)\\
&=\frac{P^{(0)}}2\frac{a'}a\hyperlink{ceres4-note}{\text{[*)]}}\left(1+\frac{2z''}{z'}+\frac{4z'''}{z'}+\text{etc.}\right).
\end{aligned}
\]

\long\def\@makefntext#1{\noindent[*)] #1}
\makeatother
\footnotetext{\hypertarget{ceres4-note}{}or \(\frac a{a'}\), the smaller in the numerator and the larger in the denominator.}
\endgroup

[For the calculation of the \(A^{(i)}\) and their derivatives, one readily finds the following formulas, which can be obtained from \textsc{Laplace}'s expansions, loc. cit., p. 270:]
\[
A^{(i)}=-P^{(i)}\qquad[\text{but}]\qquad A^{(1)}=-P'+\frac a{a'a'}
\]
\[
B^{(i)}=\tfrac12P^{(i)}-\tfrac12\left(\frac{a'}a-\frac a{a'}\right)Q^{(i)}
\]
\[
B^{(1)}=\tfrac12P'-\tfrac12\left(\frac{a'}a-\frac a{a'}\right)Q'+\frac a{a'a'}
\]
\[
C^{(i)}=-iiP^{(i)}\mathord{\text{\raisebox{-.6ex}{.}}}-\frac a{a'}Q^{(i)}
\]
\[
C^{(1)}=-P'-\frac a{a'}Q'+\frac a{a'a'}
\]
\[
\begin{gathered}
A^{(i)}+B^{(i)}+{}'B^{(i)}=0\\
B^{(i)}+C^{(i)}+{}'C^{(i)}=0.
\end{gathered}
\]
