\section*{FOURTH SECTION.}
\begin{center}
\itshape Second approximation for the latitude.
\end{center}

\par\noindent\hspace*{\parindent}In calculating the \textls[120]{latitude}, we omit from \(\omega\) and \(\pi\) the parts that
contain the factor \(\dfrac{1}{Rp}\). In the part of the third equation that contains the
perturbing forces, we substitute for \(\theta\) its value found above,
\(g\sin\lambda+\tfrac38ng(1+\tfrac14n)\sin(2E-\lambda)\), and for
\(\dfrac{\mathrm{d}\lambda}{\mathrm{d}v}\) the value \(1+\tfrac34nn\). This gives the equation
\[
\begin{aligned}
0={}&\frac{\mathrm{dd}\theta}{\mathrm{d}v^2}
+\theta\left(1+\frac32\frac{nn}{1+\mu}\frac{B^3}{R^3}
\frac{\mathrm{d}u^2}{\mathrm{d}v^2}\right)\\
&+\frac32\frac{nn}{1+\mu}g\frac{B^3}{R^3}\frac{\mathrm{d}u^2}{\mathrm{d}v^2}\Bigl\{\\
&\quad -(1+\tfrac38nn)\sin(2(v-V)-\lambda)
-\tfrac38nn\sin(2(v-V)+\lambda)\\
&\quad -\tfrac38n(1-\tfrac34n)\sin(\lambda+2nv-2V)\\
&\quad +\tfrac38nn\sin(4v-2V-2nv-\lambda)\Bigr\}.
\end{aligned}
\]

\par\noindent\hspace*{\parindent}We now have
\[
\begin{aligned}
V&=nu-2\varepsilon\sin a+\tfrac54\varepsilon\varepsilon\sin2a\\
 &=nv-2\varepsilon\sin a+\tfrac54\varepsilon\varepsilon\sin2a+2ne\sin M,
\end{aligned}
\]
\noindent where the terms of the 3\textsuperscript{rd} degree have been omitted.
Here \(a\) denotes \(nu-\mathrm{Apog.}\,\odot\), the mean anomaly of the Sun.
We shall, however, take it here to be \(nv-\mathrm{Apog.}\,\odot\), since the resulting
difference is only of the degree of the terms that we are
already neglecting. Substituting this value of \(V\) gives
\[
\begin{aligned}
2(v-V)-\lambda={}&2E-\lambda+4\varepsilon\sin a
-\tfrac52\varepsilon\varepsilon\sin2a-4ne\sin M
\end{aligned}
\]
\[
\begin{aligned}
\sin(2(v-V)-\lambda)={}&\sin(2E-\lambda)\mathbin{.}
(1-4\varepsilon\varepsilon+4\varepsilon\varepsilon\cos2a)\\
&+\cos(2E-\lambda)\mathbin{.}
(4\varepsilon\sin a-\tfrac52\varepsilon\varepsilon\sin2a-4ne\sin M)
\end{aligned}
\]
\noindent and hence our equation

apogee gives
\[
\frac{\mathrm{d}\lambda}{\mathrm{d}v}
=\BareRoot{\left\{1+\tfrac32\frac{nn}{1+\mu}
(1+2ee+\tfrac32\varepsilon\varepsilon-\tfrac12gg-\tfrac38n-\tfrac{79}{32}nn)\right\}},

\[
\left[\frac{\mathrm{d}\lambda}{\mathrm{d}v}\right]
=1+\tfrac34\frac{nn}{1+\mu}
