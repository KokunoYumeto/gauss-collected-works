\begin{center}\pdfbookmark[2]{5. Integration of the perturbation equations}{ceres-005}[5.]\end{center}
\noindent\hspace*{\parindent}[The right-hand sides of equations 11 and 12 for \(\delta r\) and \(\delta v\) can be calculated as follows.] Let [in general]
\[
\text{a term of }Q=A\cos\{(kn-k'n')t-a\}=A\cos\psi,
\]
\noindent[and set] approximately [that is, neglecting the 2nd powers of the eccentricities:]
\[V=M-2e\sin M,\]
\[\sin V=\sin M-e\sin2M,\]
\[\cos V=\cos M+e-e\cos2M,\]
\[r=a(1+e\cos M),\]
\[r\sin V=a(\sin M-\tfrac12e\sin2M),\]
\[r\cos V=a(\cos M+\tfrac32e-\tfrac12e\cos2M),\]
\noindent then [the corresponding \hypertarget{native-393}{}term in] \(rQ\sin V\) is
\[
\begin{aligned}
={}&\tfrac12aA\{\sin(M+\psi)+\sin(M-\psi)\}\\
&-\tfrac14eaA\{\sin(2M+\psi)+\sin(2M-\psi)\}
\end{aligned}
\]
\noindent term in] \(rQ\cos V\)
\[
\begin{aligned}
={}&\tfrac12aA\{\cos(M+\psi)+\cos(M-\psi)\}+\tfrac32eaA\cos\psi\\
&-\tfrac14eaA\{\cos(2M+\psi)+\cos(2M-\psi)\}.
\end{aligned}
\]

After integration
\[
\begin{gathered}
{}[\text{term in }\cos V\int rQ\sin V\mathord{.}\,n\mathrm{d}t=]\\[1ex]
\left\{\begin{aligned}
&-\frac{\tfrac12aA\cos(M+\psi)}{1+k-k'\theta}-\frac{\tfrac12aA\cos(M-\psi)}{1-k+k'\theta}\\
&+\frac{\tfrac14eaA\cos(2M+\psi)}{2+k-k'\theta}+\frac{\tfrac14eaA\cos(2M-\psi)}{2-k+k'\theta}
\end{aligned}\right\}\\[1ex]
(\cos M+e-e\cos2M)
\end{gathered}
\]
\[
\begin{gathered}
{}[\text{term in }-\sin V\int rQ\cos V\mathord{.}\,n\mathrm{d}t=]\\[1ex]
\left\{\begin{aligned}
&-\frac{\tfrac12aA\sin(M+\psi)}{1+k-k'\theta}-\frac{\tfrac12aA\sin(M-\psi)}{1-k+k'\theta}\\
&-\frac{\tfrac32eaA\sin\psi}{k-k'\theta}\\
&+\frac{\tfrac14eaA\sin(2M+\psi)}{2+k-k'\theta}+\frac{\tfrac14eaA\sin(2M-\psi)}{2-k+k'\theta}
\end{aligned}\right\}\\[1ex]
(\sin M-e\sin2M).
\end{gathered}
\]

From this follows [by 11] the corresponding
\[
\begin{gathered}
{}[14]\quad\text{term in }\delta r=\\[1ex]
\frac{\mu aaA}{\BareRoot{(1-ee)}}\left\{\begin{aligned}
&-\frac{\cos\psi}{1-(k-k'\theta)^2}+\frac{\tfrac14e\cos(M+\psi)}{2+k-k'\theta}\\
&+\frac{(k-k'\theta)e\cos(M+\psi)}{1-(k-k'\theta)^2}+\frac{\tfrac34e\cos(M+\psi)}{k-k'\theta}\\
&+\frac{\tfrac14e\cos(M-\psi)}{2-k+k'\theta}-\frac{(k-k'\theta)e\cos(M-\psi)}{1-(k-k'\theta)^2}\\
&-\frac{\tfrac34e\cos(M-\psi)}{k-k'\theta}
\end{aligned}\right\}
\end{gathered}
\]

[This formula has an exception when one of the denominators vanishes, first when \(\psi=0\) and thus \(k=k'=0\). In that case substitute
\[
-\frac{\mu aaA}{\BareRoot{(1-ee)}}(1-\tfrac14e\cos M+\tfrac32ent\sin M);
\]
\noindent and second,] for \(\psi=nt-\text{constant}\) [so that \(k=1\), \(k'=0\)], [this term] becomes
\[
\begin{gathered}
=\frac{\mu aaA}{\BareRoot{(1-ee)}}\\[1ex]
\left\{\begin{aligned}
&-\tfrac14\cos\psi-\tfrac12nt\sin\psi\\
&\left[\begin{aligned}
&+\tfrac7{12}e\cos(M+\psi)+\tfrac12ent\sin(M+\psi)\\
&-\tfrac14e\cos(M-\psi)+\tfrac12ent\sin(M-\psi)
\end{aligned}\right]
\end{aligned}\right\}
\end{gathered}
\]
\noindent[where \(M-\psi\) is a constant.

Expression 12 for \(\delta v\) can be written as follows:]
\[
\begin{aligned}
{}[15]\quad\delta v=\frac1{\BareRoot{(1-ee)}}\Bigg\{&\frac{2\BareRoot{\frac ra}\,\mathrm{d}\left(\rho\BareRoot{\frac ra}\right)}{an\mathrm{d}t}\\
&+2\mu a\int Qn\mathrm{d}t-\mu a\iint\partial Rn\mathrm{d}t\Bigg\}
\end{aligned}
\]

[First calculate the first term in the braces:] the contribution of [the part of \(Q\) equal to] \(A\cos\psi\) to \(\rho\) \hypertarget{native-394}{}[is to be taken from equation 14. Multiplying it by] \(\BareRoot{\frac ra}=1+\tfrac12e\cos M\) [gives the corresponding part of \(\frac\rho a\BareRoot{\frac ra}\)]=
\[
\begin{gathered}
\frac{\mu aA}{\BareRoot{(1-ee)}}\\[1ex]
\left\{\begin{aligned}
&-\frac{\cos\psi}{1-(k-k'\theta)^2}\\
&+\left(\frac{\tfrac14}{2+k-k'\theta}+\frac{k-k'\theta-\tfrac14}{1-(k-k'\theta)^2}+\frac{\tfrac34}{k-k'\theta}\right)e\cos(M+\psi)\\
&+\left(\frac{\tfrac14}{2-k+k'\theta}-\frac{k-k'\theta+\tfrac14}{1-(k-k'\theta)^2}-\frac{\tfrac34}{k-k'\theta}\right)e\cos(M-\psi)
\end{aligned}\right\},
\end{gathered}
\]
\noindent hence [the corresponding part of] \(2\frac{\mathrm{d}\left(\frac\rho a\BareRoot{\frac ra}\right)}{n\mathrm{d}t}=\)
\[
\begin{gathered}
\frac{\mu aA}{\BareRoot{(1-ee)}}\\[1ex]
\left\{\begin{aligned}
&\frac{2(k-k'\theta)}{1-(k-k'\theta)^2}\sin\psi\\
&-\left(\frac{\tfrac12(1+k-k'\theta)}{2+k-k'\theta}+\frac{2(k-k'\theta)-\tfrac12}{1-k+k'\theta}+\frac{\tfrac32(1+k-k'\theta)}{k-k'\theta}\right)\\
&\hfill e\sin(M+\psi)\\
&-\left(\frac{\tfrac12(1-k+k'\theta)}{2-k+k'\theta}-\frac{2(k-k'\theta)+\tfrac12}{1+k-k'\theta}-\frac{\tfrac32(1-k+k'\theta)}{k-k'\theta}\right)\\
&\hfill e\sin(M-\psi)
\end{aligned}\right\}.
\end{gathered}
\]

[It follows that the corresponding part of \(\frac{2\BareRoot{\frac ra}\,\mathrm{d}\left(\rho\BareRoot{\frac ra}\right)}{an\mathrm{d}t}\) equals the preceding expression, to which] multiplication by \(\BareRoot{\frac ra}\) adds the following on the right-hand side:
\[
\frac{\mu aA}{\BareRoot{(1-ee)}}\left\{\begin{aligned}
&\tfrac12\frac{k-k'\theta}{1-(k-k'\theta)^2}e\sin(M+\psi)\\
&-\tfrac12\frac{k-k'\theta}{1-(k-k'\theta)^2}e\sin(M-\psi)
\end{aligned}\right\}.
\]

[The last two expressions again have an exception, first when \(\psi=\text{constant}\). The term in \(\frac{2\BareRoot{\frac ra}\,\mathrm{d}\left(\rho\BareRoot{\frac ra}\right)}{an\mathrm{d}t}\) then becomes
\[
-\frac{\mu aA}{\BareRoot{(1-ee)}}\{\tfrac52e\sin M+3entM\}
\]
\noindent and second,] for \(\psi=nt-\text{constant}\) [the calculation gives this term in] \(\frac{2\BareRoot{\frac ra}\,\mathrm{d}\left(\rho\BareRoot{\frac ra}\right)}{an\mathrm{d}t}=\)
\[
\begin{gathered}
\frac{\mu aA}{\BareRoot{(1-ee)}}\\[1ex]
\left\{\begin{aligned}
&-\tfrac12\sin\psi-nt\cos\psi\\
&\left[\begin{aligned}
&-\tfrac{35}{24}e\sin(M+\psi)+\tfrac{11}8e\sin(M-\psi)\\
&+\tfrac54ent\cos(M+\psi)-\tfrac14ent\cos(M-\psi)
\end{aligned}\right]
\end{aligned}\right\}.
\end{gathered}
\]
\noindent[The rest of \(\delta v\) is readily formed.]
