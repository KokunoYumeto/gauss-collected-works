As an example, we calculate the longitude and latitude from the right ascension $355^0\,43'\,45''{,}30=\alpha$, the declination $-8^0\,47'\,25''=\delta$, and the obliquity of the ecliptic $23^0\,27'\,59''{,}26=\epsilon$. Thus $45^0+\tfrac12\alpha=222^0\,51'\,52''{,}65$, $45^0-\tfrac12(\epsilon+\delta)=37^0\,39'\,42''{,}87$, $45^0-\tfrac12(\epsilon-\delta)=28^0\,52'\,17''{,}87$, giving
\begin{center}
\begin{tabular}{@{}ll@{}}
\makebox[21em][l]{$\log\cos(45^0+\tfrac12\alpha)$\hfill .\,.\,.\,.\,.\,.\,.} & $9{,}865\,0820_n$\\[.3em]
\makebox[21em][l]{$\log\sin(45^0-\tfrac12(\epsilon+\delta))$\hfill .\,.\,.} & $9{,}786\,0418$\\[.3em]
\makebox[21em][l]{$\log\cos(45^0-\tfrac12(\epsilon+\delta))$\hfill .\,.\,.} & $9{,}898\,5222$\\[.3em]
\makebox[21em][l]{$\log\sin(45^0+\tfrac12\alpha)$\hfill .\,.\,.\,.\,.\,.\,.} & $9{,}832\,6803_n$\\[.3em]
\makebox[21em][l]{$\log\sin(45^0-\tfrac12(\epsilon-\delta))$\hfill .\,.\,.} & $9{,}683\,8112$\\[.3em]

\noindent from which we obtain $\tfrac12(E-l)=216^0\,56'\,5''{,}39$ and $\log\sin(45^0-\tfrac12 b)=9{,}872\,3171$.
\begin{center}
\begin{tabular}{@{}ll@{}}
\makebox[24em][l]{$\log\cos(45^0-\tfrac12 b)\sin\tfrac12(E+l)$\hfill .\,.\,.} & $9{,}516\,4915_n$\\[.3em]
\makebox[24em][l]{$\log\cos(45^0-\tfrac12 b)\cos\tfrac12(E+l)$\hfill .\,.\,.} & $9{,}763\,6042_n$\\[.3em]
\end{tabular}
\end{center}
\noindent from which we obtain $\tfrac12(E+l)=\grave{2}09^0\,30'\,49''{,}94$ and $\log\cos(45^0-\tfrac12 b)=9{,}823\,9669$.

\SourceLeaf{089}We thus obtain $E=426^0\,26'\,55''{,}33$, $l=-7^0\,25'\,15''{,}45$, or equivalently $E=66^0\,26'\,55''{,}33$, $l=352^0\,34'\,44''{,}55$. For the angle $45^0-\tfrac12 b$ we obtain $48^0\,10'\,58''{,}11$ from the logarithm of the sine, $48^0\,10'\,58''{,}17$ from the logarithm of the cosine, and $48^0\,10'\,58''{,}14$ from the tangent, whose logarithm is the difference of the other two logarithms. Hence $b=-6^0\,21'\,56''{,}28$.
