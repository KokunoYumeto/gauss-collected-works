\begin{center}3.\end{center}
\noindent\hspace*{\parindent}Hieraus folgt, dass für \(x=1\)
\[
\begin{aligned}
\xi&=c\\
\xi'&=-c+1\\
\xi''&=+3c-2\\
\xi'''&=-13c+8\\
\xi^{\mathrm{IV}}&=+73c-44\\
\xi^{\mathrm{V}}&=-501c+300\\
\xi^{\mathrm{VI}}&=+4051c-2420\\
\xi^{\mathrm{VII}}&=-37633c+22460.
\end{aligned}
\]
\begin{center}4.\end{center}
\noindent\hspace*{\parindent}Mithin der Werth von \(\xi\) für \(x=1-\omega\)
\begin{equation}
\left\{\begin{aligned}
&c\left(1+\omega+\frac32\omega^2+\frac{13}{6}\omega^3+\frac{73}{24}\omega^4+\frac{501}{120}\omega^5+\text{etc.}\right)\\[.5em]
&-\left(\omega+\omega^2+\frac86\omega^3+\frac{44}{24}\omega^4+\frac{300}{120}\omega^5+\frac{2420}{720}\omega^6+\text{etc.}\right)
\end{aligned}\right.\tag*{[3]}
\end{equation}
\noindent Diese Reihen convergiren, wenn \(\omega<1\).
\begin{center}5.\end{center}
\noindent\hspace*{\parindent}Endlich ist
\begin{equation}
c=\frac{\omega+\omega\omega+\frac86\omega^3+\text{etc.}}{1+\omega+\frac32\omega^2+\text{etc.}}\tag*{[4]}
\end{equation}
\begin{equation}

\end{equation}
