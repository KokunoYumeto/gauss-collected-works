\NumberedItem{2}
\par\indent Zahlen aus zwei Factoren
\[
\frac{lla.a}{la},
\]
\noindent (wahrsch.) aus 3 Factoren
\[
\frac{\frac12(lla)^2a}{la},\ .\,.\,.
\]
\noindent et sic in inf.
\ClosingRule
