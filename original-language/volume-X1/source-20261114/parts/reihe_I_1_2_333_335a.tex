\[\begin{gathered}
1=P-\alpha Q\\
P=Q-\beta R\\
Q=R-\gamma S\\
R=S-\delta T\\

\end{gathered}\]
\[P=\cfrac1{1-\alpha\frac QP}
=\cfrac1{1-\cfrac\alpha{1-\frac{\beta R}Q}}
=\cfrac1{1-\cfrac\alpha{1-\cfrac\beta{1-\frac{\gamma S}R}}}\quad\text{etc.}\]
