\begin{equation*}\tag*{[7]}
\left\{\begin{aligned}
&(1+2x+2x^4+2x^9+{.}{.}{.})+(1+2x^3+2x^{12}+2x^{27}+{.}{.}{.})\\
&=2(1+x)(1+x^3)(1+x^5)(1+x^7){.}{.}{.}\\
&\qquad\times(1-x^5)(1-x^7)(1-x^{17})(1-x^{19}){.}{.}{.}\\

&=\frac{2\TransBracket{xx}^2\TransBracket{x^{12}}}{\TransBracket{x}\TransBracket{x^4}}
