\par\indent Aequationis
\[\frac{dx}{\surd(1+\mu\mu xx)(1-xx)}+\frac{dy}{\surd(1+\mu\mu yy)(1-yy)}=0\]
\noindent integrale completum est
\[\frac{y\surd(1+\mu\mu xx)(1-xx)+x\surd(1+\mu\mu yy)(1-yy)}{1+\mu\mu xxyy}=\text{Const.}\]

\noindent Statuendo itaque
\[\int\frac{dx}{\surd(1+\mu\mu xx)(1-xx)}=u\]
\noindent atque
\[\begin{aligned}
x&=Fu\\1&=F\tfrac12\mathrm{II}\\y&=Fw\\
x'&=F(\tfrac12\mathrm{II}-u)\\y'&=F(\tfrac12\mathrm{II}-w)
\end{aligned}\]
\noindent fit
\begin{equation*}\tag*{1)}
\frac{y\sqrt{1+\mu\mu xx.1-xx}+x\sqrt{1+\mu\mu yy.1-yy}}{1+\mu\mu xxyy}=F\overline{u+w}.
\end{equation*}
\noindent Quare faciendo \(y=1\) fit
\[\surd\frac{1-xx}{1+\mu\mu xx}=F(\tfrac12\mathrm{II}+u)=F(\tfrac12\mathrm{II}-u)=Gu=x'\]
\noindent atque
\[1=xx+x'x'+\mu\mu xx x'x'\quad.\quad.\quad.\quad\mathrm I\]
\[\frac{yx'+xy'+\mu\mu xx'(xy+x'y')}{1+\mu\mu xxyy}=F\overline{u+w}.\]
