[4. Zu Art. 76–77.]\par
Allgemeine Differentialformeln für die geocentrischen Örter der Planeten.
\end{center}
\noindent
\(l\) geocentrische Länge des Planeten,\\
\hfill \(l-\AscendingNode=\eta\)\\
\(L\) heliocentrische Länge der Erde\\
\(u\) Argument der Breite\\
\(b\) geocentrische Breite\\{}
[\(v\) wahre Anomalie\\
\(\tilde\omega\) Länge des Perihels\\
\(\DailyMotionSign\) mittlere tägliche Bewegung\\
Ep. mittlere Länge für die Epoche\\
\(n\) seit der Epoche verflossene Zeit in Tagen.]
\[
\begin{aligned}
1.\quad&a\operatorname{tang}\varphi\sin v=\alpha=\left(\frac{\mathrm{d}r}{\mathrm{d}M}\right)\\
2.\quad&-a\cos\varphi\cos v=\alpha'=\left(\frac{\mathrm{d}r}{\mathrm{d}\varphi}\right)\\
3.\quad&\frac{aa\cos\varphi}{rr}=\beta=\left(\frac{\mathrm{d}v}{\mathrm{d}M}\right)\\
4.\quad&\frac{aa\sin E}{rr}(2-ee-e\cos E)=\beta'=\left(\frac{\mathrm{d}v}{\mathrm{d}\varphi}\right)
\end{aligned}
\]
\[
5.\quad\frac{\operatorname{tang}\eta}{\cos i}=\operatorname{tang}M
\]
\[
\begin{aligned}
6.\quad&\sin\eta\operatorname{tang}i=\operatorname{tang}N;\\
&\cos N=\frac{\cos M}{\cos\eta}\mathpunct{\raisebox{.45ex}{,}}\quad\sin N=\sin M\sin i
\end{aligned}
\]
\[
\begin{aligned}
7.\quad&\frac{\operatorname{tang}(M-u)}{\cos\eta\sin i}=\operatorname{tang}P\\
8.\quad&\frac{R}{\Delta'}\sin(L-l)=x
\end{aligned}
\]

\[
\begin{aligned}
9.\quad&\frac{x}{r}=A=\left(\frac{\mathrm{d}l}{\mathrm{d}r}\right)\\
10.\quad&-x\operatorname{cotang}(M-u)=B=\left(\frac{\mathrm{d}l}{\mathrm{d}u}\right)\\
11.\quad&-\cos\eta\operatorname{tang}b=C=\left(\frac{\mathrm{d}l}{\mathrm{d}i}\right)\\
12.\quad&\frac{R}{\Delta'}\cos(L-l)=y\\
13.\quad&1+y=D=\left(\frac{\mathrm{d}l}{\mathrm{d}\AscendingNode}\right)
\end{aligned}
\]
\[
\begin{aligned}
14.\quad&-\frac{y\sin b\cos b}{r}=A'=\left(\frac{\mathrm{d}b}{\mathrm{d}r}\right)\\
15.\quad&\frac{r\sin(M-u)\cos(N-b-P)}{\Delta\sin P}=B'=\left(\frac{\mathrm{d}b}{\mathrm{d}u}\right)\\
16.\quad&\frac{r\sin u\cos i\cos(N-b)}{\Delta\cos N}=C'=\left(\frac{\mathrm{d}b}{\mathrm{d}i}\right)\\
17.\quad&x\sin b\cos b=D'=\left(\frac{\mathrm{d}b}{\mathrm{d}\AscendingNode}\right)
\end{aligned}
\]
\[
\begin{aligned}
18.\quad&A\alpha+B\beta=E\\
19.\quad&A'\alpha+B'\beta=E'\\
20.\quad&A\alpha'+B\beta'=F\\
21.\quad&A'\alpha'+B'\beta'=F'
\end{aligned}
\]
\[
\begin{aligned}
\mathrm{d}l={}&E\,\mathrm{d}\mathrm{Ep.}+\left(nE-\frac{2rA}{3\DailyMotionSign}\right)\mathrm{d}\DailyMotionSign+(B-E)\,\mathrm{d}\tilde\omega\\
&+F\,\mathrm{d}\varphi+C\,\mathrm{d}i+(D-B)\,\mathrm{d}\AscendingNode
\end{aligned}
\]
\[
\begin{aligned}
\mathrm{d}b={}&E'\,\mathrm{d}\mathrm{Ep.}+\left(nE'-\frac{2rA'}{3\DailyMotionSign}\right)\mathrm{d}\DailyMotionSign+(B'-E')\,\mathrm{d}\tilde\omega\\
&+F'\,\mathrm{d}\varphi+C'\,\mathrm{d}i+(D'-B')\,\mathrm{d}\AscendingNode.
\end{aligned}
\]
