\mathrm{II}\pdot\pdot\pdot\;G,\quad

\[

4\Theta&=R=0\\
4\Theta'&=-g+G-h-H+R'\\
4\Theta''&=-2g+2G+R''\\

\end{aligned}
\]

\[
\begin{gathered}
G=0,\quad g=-1,\quad H=-1,\quad h=+2,\\
